# Python packages installed into the miniforge BASE environment.
#
# The TDBI grid runs on ~/miniforge3/bin/python3 (base), not on a named environment, so its
# dependencies belong here. Installed by 3_deploy.sh; verified by 4_test.sh.
#
# The citizens themselves are stdlib-only by design -- herald, orchestrator, linter, and
# registrar have no dependencies at all, so the grid runs on a bare Python. Everything below is
# for a TOOL that sits beside the citizens: retrieval (ADR-0033) and the corpus web harness
# (ADR-0039). A session never needs any of them, which is why a failure here costs search or the
# harness and never the ritual.

# ONNX embedding runtime. NOT PyTorch: the same models (nomic, BGE, MiniLM) at ~50MB instead of
# ~2GB. On a client laptop that is the difference between a trivial install and a security
# conversation -- and zero-procurement deployability is a standing constraint, not a preference.
fastembed>=0.8.0

# Vector store: one sqlite file, no server, no account, SQL filters over metadata.
sqlite-vec>=0.1.9

# The corpus web harness (TDBI tools/harness/server.py, ADR-0039). It has imported flask since it
# was written and nothing installed it, so the harness has never once been runnable from a clean
# bootstrap -- it worked only on machines where flask arrived as some other package's dependency.
# That is the whole failure mode a declared base-package list exists to prevent.
flask>=3.0.0

# Data analysis and document tooling used by corpus tooling and recology workspace scripts.
pdfplumber>=0.11.0
matplotlib>=3.9.0
openpyxl>=3.1.0
